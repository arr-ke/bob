    id INT AUTO_INCREMENT PRIMARY KEY,
    name1 varchar(100) NOT NULL,
    name2 varchar(100) NOT NULL,
    name3 varchar(100) NOT NULL,
    name4 varchar(100) NOT NULL,
    name5 varchar(100) NOT NULL
);
